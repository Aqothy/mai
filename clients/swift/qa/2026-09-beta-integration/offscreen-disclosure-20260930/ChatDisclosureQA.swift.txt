#if os(macOS) && DEBUG
import AppKit
import SwiftUI
import Observation
import Synchronization

@MainActor enum ChatDisclosureQA {
struct DisclosureEvent { let kind: String; let expanded: Bool }
static weak var qaWindow: NSWindow?
static var disclosureEvents: [DisclosureEvent] = []
static func recordDisclosure(_ kind: String, expanded: Bool) {
    guard isRequested else { return }
    disclosureEvents.append(DisclosureEvent(kind: kind, expanded: expanded))
    qaWindow?.title = "QA measuring disclosure"
}
static var requestURL: URL { FileManager.default.temporaryDirectory.appending(path: "mai-disclosure-qa-request.json") }
static var isRequested: Bool { FileManager.default.fileExists(atPath: requestURL.path) }
static func run() async {
// Run the normal app via Xcode MCP on My Mac; capture-window.py takes checkpoints.
// A separate accessibility client activates actual disclosure controls.
do {
try await Task { @MainActor in
    let request = (try? Data(contentsOf: requestURL)).flatMap { try? JSONDecoder().decode([String: String].self, from: $0) } ?? [:]
    let output = URL(fileURLWithPath: request["output"] ?? UserDefaults.standard.string(forKey: "ChatDisclosureQAOutput") ?? FileManager.default.temporaryDirectory.appending(path: "qa-switch-fallback").path)
    let renderer = request["renderer"] ?? (UserDefaults.standard.bool(forKey: "ChatBenchmarkUseList") ? "list" : "native")
    print("DISCLOSURE_QA_RUNTIME \(ProcessInfo.processInfo.operatingSystemVersionString) pid=\(ProcessInfo.processInfo.processIdentifier)")
    try FileManager.default.createDirectory(at: output, withIntermediateDirectories: true)
    func require(_ condition: Bool, _ message: String) throws {
        if !condition { throw NSError(domain: "DisclosureQA", code: 1, userInfo: [NSLocalizedDescriptionKey: message]) }
    }
    func save(_ name: String, _ value: Any) throws {
        try JSONSerialization.data(withJSONObject: value, options: [.prettyPrinted, .sortedKeys])
            .write(to: output.appending(path: name + ".json"), options: .atomic)
    }
    @MainActor final class FixtureRPC: ThreadRPCClient {
        nonisolated deinit {}
        var onNotification: ((String, Data) -> Void)?
        var onDisconnect: (((any Error)?) -> Void)?
        let threads: [Thread]
        init(_ threads: [Thread]) { self.threads = threads }
        func connect() {}
        func disconnect() {}
        func subscribeThreadList() async throws -> ThreadListStreamItem {
            .init(kind: "snapshot", sequence: nil,
                  snapshot: .init(snapshotSequence: 0, threads: threads.map(ChatSyntheticBenchmarkThread.listEntry), updatedAt: .now), thread: nil)
        }
        func subscribeThread(_ input: SubscribeThreadInput) async throws -> ThreadStreamItem {
            guard let thread = threads.first(where: { $0.id == input.threadID }) else { throw URLError(.badURL) }
            return .init(event: nil, kind: "snapshot", snapshot: .init(historyRestorePending: false, snapshotSequence: 1, thread: thread))
        }
        func unsubscribeThread(_ input: SubscribeThreadInput) async throws {}
        func send(_ event: Event) throws {
            struct Notification: Encodable { let params: ThreadStreamItem }
            let item = ThreadStreamItem(event: event, kind: "event", snapshot: nil)
            onNotification?(MaidRPCMethod.orchestrationSubscribeThread, try newJSONEncoder().encode(Notification(params: item)))
        }
    }
    let oldArguments = UserDefaults.standard.volatileDomain(forName: UserDefaults.argumentDomain)
    var arguments = oldArguments
    arguments["ChatBenchmarkUseList"] = renderer == "list"
    UserDefaults.standard.setVolatileDomain(arguments, forName: UserDefaults.argumentDomain)
    defer { UserDefaults.standard.setVolatileDomain(oldArguments, forName: UserDefaults.argumentDomain) }
    let suite = "mai-switch-qa-" + UUID().uuidString
    guard let defaults = UserDefaults(suiteName: suite) else { throw URLError(.unknown) }
    defer { defaults.removePersistentDomain(forName: suite) }
    let original = ChatSyntheticBenchmarkThread.thread(turnCount: 20, identity: "qa-chat-a")
    let a = original.with(timeline: original.timeline.map { entry in
        if let message = entry.message, message.role == "assistant" {
            let number = Int(message.turnID?.split(separator: "-").last ?? "0") ?? 0
            let text = "ANSWER \(number + 1) — " + String(repeating: "This completed answer remains in place while earlier activity opens. Keep the reader's position and the streamed reply source intact. ", count: 4)
            let date = message.createdAt.addingTimeInterval(Double(number - 22))
            return entry.with(message: message.with(createdAt: date, text: text, updatedAt: date))
        }
        if let item = entry.item, item.kind == MaidItemKind.reasoning.rawValue {
            return entry.with(item: item.with(payload: JSONAny(["text": String(repeating: "Thought detail: compare the alternatives, preserve exact content, and verify stable reading positions. ", count: 20)])))
        }
        return entry
    }, title: "DISCLOSURE QA")
    let originalAIDs = a.timeline.map { $0.message?.id ?? $0.item?.id ?? "missing" }
    let rpc = FixtureRPC([a])
    let store = ThreadStore(rpc: rpc)
    await store.start()
    store.selectThread(a.id)
    let qaScrollState = ChatScrollState()
    let host = NSHostingView(rootView: ChatView(store: store, draftStore: ThreadDraftStore(defaults: defaults), qaScrollState: qaScrollState, openThread: nil))
    host.sizingOptions = []
    let policy = NSApplication.shared.activationPolicy()
    NSApplication.shared.setActivationPolicy(.regular)
    defer { NSApplication.shared.setActivationPolicy(policy) }
    let window = NSWindow(contentRect: NSRect(x: 120, y: 100, width: 1100, height: 800), styleMask: [.titled, .resizable], backing: .buffered, defer: false)
    qaWindow = window
    defer { qaWindow = nil }
    window.isReleasedWhenClosed = false
    window.title = "mai scroll intent QA " + renderer
    window.contentView = host
    window.level = .floating
    window.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary]
    NSApplication.shared.unhide(nil)
    NSApplication.shared.activate()
    window.makeKeyAndOrderFront(nil)
    window.orderFrontRegardless()
    defer { window.orderOut(nil); window.close() }
    var sequence = 1000
    @MainActor func send(_ type: MaidEventType, messageID: String? = nil, text: String? = nil, session: SessionBinding? = nil) throws {
        sequence += 1
        let payload = EventPayload(approval: nil, attachments: nil, configOptions: nil, createdAt: nil, cwd: nil, decision: nil,
            item: nil, messageID: messageID, modelSelection: nil, optionID: nil, plan: nil, providerInstanceID: nil,
            requestID: nil, role: messageID == nil ? nil : "assistant", session: session, sessionCleared: nil,
            slashCommands: nil, stopReason: nil, text: text, threadID: a.id, title: nil, tokenUsage: nil,
            turnID: "qa-stream-turn", updatedAt: nil, value: nil)
        try rpc.send(Event(actor: nil, commandID: nil, eventID: "qa-event-\(sequence)", metadata: nil, occurredAt: .now,
                           payload: payload, sequence: sequence, type: type.rawValue))
    }
    @MainActor func select(_ id: String) async throws {
        store.selectThread(id)
        let deadline = ContinuousClock.now + .seconds(5)
        while store.selectedThread?.id != id && ContinuousClock.now < deadline { try await Task.sleep(for: .milliseconds(20)) }
        try require(store.selectedThread?.id == id, "Selected chat failed to load")
        try await Task.sleep(for: .milliseconds(500))
    }
    var checkpoints: [[String: Any]] = []
    @MainActor func capture(_ name: String) async throws {
        host.layoutSubtreeIfNeeded()
        let scroll = ChatBenchmarkModel.transcriptScrollView(in: window)
        let value: [String: Any] = ["name": name, "windowID": window.windowNumber,
            "selected": store.selectedThreadID ?? "none", "renderer": renderer,
            "visible": window.occlusionState.contains(.visible), "width": host.bounds.width,
            "shouldFollowBottom": qaScrollState.shouldFollowBottom, "isNearBottom": qaScrollState.isNearBottom,
            "clipX": scroll?.contentView.bounds.minX ?? -1, "clipY": scroll?.contentView.bounds.minY ?? -1,
            "clipWidth": scroll?.contentView.bounds.width ?? -1, "documentHeight": scroll?.documentView?.bounds.height ?? -1,
            "streamCharacters": store.cachedThread(for: a.id)?.timeline.last?.message?.text.count ?? -1,
            "unixTime": Date.now.timeIntervalSince1970]
        checkpoints.append(value)
        try require(window.occlusionState.contains(.visible), "Window is not visible")
        try save("capture-request", value)
        let deadline = ContinuousClock.now + .seconds(5)
        while !FileManager.default.fileExists(atPath: output.appending(path: name + ".done").path) && ContinuousClock.now < deadline {
            try await Task.sleep(for: .milliseconds(20))
        }
        try require(FileManager.default.fileExists(atPath: output.appending(path: name + ".done").path), "Window capture missing: " + name)
    }
    var stages: [[String: Any]] = []
    @MainActor func transcript() throws -> NSScrollView {
        guard let scroll = ChatBenchmarkModel.transcriptScrollView(in: window), scroll.documentView != nil else { throw URLError(.unknown) }
        return scroll
    }
    var sent = ""
    var finishStreaming = false
    @MainActor func stage(_ name: String) async throws {
        let scroll = try transcript()
        let startY = scroll.contentView.bounds.minY
        let startHeight = scroll.documentView?.bounds.height ?? 0
        guard let document = scroll.documentView as? any ChatMacScrollDocument else { throw URLError(.unknown) }
        let visibleRows = document.rows(in: scroll.contentView.documentVisibleRect)
        let anchor = (visibleRows.location..<NSMaxRange(visibleRows)).first { document.rect(ofRow: $0).height > 1 } ?? visibleRows.location
        let anchorBefore = document.rect(ofRow: anchor).minY - scroll.contentView.documentVisibleRect.minY
        let startChars = sent.count
        try save("stage", ["name": name, "originY": startY, "documentHeight": startHeight,
            "characters": startChars, "following": qaScrollState.shouldFollowBottom])
        let deadline = ContinuousClock.now + .seconds(180)
        let eventStart = disclosureEvents.count
        let expectedKind = String(name.split(separator: "-")[1])
        let expectedExpanded = name.hasSuffix("open")
        @MainActor func hasExpectedEvent() -> Bool {
            disclosureEvents.dropFirst(eventStart).contains { $0.kind == expectedKind && $0.expanded == expectedExpanded }
        }
        window.title = "QA ready " + name
        var originSamples: [[String: Any]] = []
        while !hasExpectedEvent() && ContinuousClock.now < deadline {
            let current = try transcript()
            originSamples.append(["time": Date.now.timeIntervalSince1970, "originY": current.contentView.bounds.minY,
                "height": current.documentView?.bounds.height ?? -1, "following": qaScrollState.shouldFollowBottom,
                "document": current.documentView.map { String(reflecting: type(of: $0)) } ?? "nil"])
            if originSamples.count % 10 == 0 { try save(name + "-origins", originSamples) }
            try await Task.sleep(for: .milliseconds(30))
        }
        try save(name + "-origins", originSamples)
        try require(hasExpectedEvent(), "Missing actual disclosure action for " + name)
        try await Task.sleep(for: .milliseconds(400))
        let endY = scroll.contentView.bounds.minY
        let height = scroll.documentView?.bounds.height ?? 0
        let anchorAfter = document.rect(ofRow: anchor).minY - scroll.contentView.documentVisibleRect.minY
        let result: [String: Any] = ["name": name, "originBefore": startY, "originAfter": endY,
            "heightBefore": startHeight, "heightAfter": height, "charactersBefore": startChars,
            "charactersAfter": sent.count, "following": qaScrollState.shouldFollowBottom,
            "nearBottom": qaScrollState.isNearBottom, "anchorRow": anchor,
            "anchorBefore": anchorBefore, "anchorAfter": anchorAfter]
        stages.append(result)
        try save("stages", stages)
        try await capture(name)
        try require(abs(anchorAfter - anchorBefore) <= 1, "Disclosure moved the reading anchor: " + name)
        try require(!qaScrollState.shouldFollowBottom, "Disclosure unexpectedly resumed following: " + name)
        try require(sent.count > startChars, "Streaming did not advance during " + name)
        // Stream growth also contributes to total document height. Separate
        // controls/visible content are verified through accessibility and captures.
    }
    do {
        try await select(a.id)
        let deadline = ContinuousClock.now + .seconds(8)
        while ChatBenchmarkModel.transcriptScrollView(in: window) == nil && ContinuousClock.now < deadline { try await Task.sleep(for: .milliseconds(50)) }
        try await Task.sleep(for: .seconds(1))
        try await capture("initial")
        try save("stage", ["name": "accessibility-setup"])
        let setupDeadline = ContinuousClock.now + .seconds(180)
        while !FileManager.default.fileExists(atPath: output.appending(path: "accessibility-setup.input-done").path) && ContinuousClock.now < setupDeadline { try await Task.sleep(for: .milliseconds(30)) }
        try require(FileManager.default.fileExists(atPath: output.appending(path: "accessibility-setup.input-done").path), "Accessibility setup missing")
        qaScrollState.requestScrollToBottom()
        try await Task.sleep(for: .seconds(1))
        try await capture("ready")
        let scroll = try transcript()
        let staticBottom = scroll.contentView.bounds.minY
        var probes: [[String: Any]] = []
        @MainActor func probe(_ name: String) throws {
            let current = try transcript()
            probes.append(["name": name, "requestedStaticBottom": staticBottom,
                "oldID": String(describing: ObjectIdentifier(scroll)), "currentID": String(describing: ObjectIdentifier(current)),
                "oldDocument": String(describing: type(of: scroll.documentView)),
                "currentDocument": String(describing: type(of: current.documentView)),
                "oldOrigin": scroll.contentView.bounds.minY, "currentOrigin": current.contentView.bounds.minY,
                "oldHeight": scroll.documentView?.bounds.height ?? -1, "currentHeight": current.documentView?.bounds.height ?? -1,
                "oldWindow": scroll.window?.windowNumber ?? -1, "currentWindow": current.window?.windowNumber ?? -1,
                "following": qaScrollState.shouldFollowBottom])
            try save("position-probes", probes)
        }
        try probe("before-stream")
        try send(.threadTurnStartRequested)
        sent = String(repeating: "Initial streamed content keeps the activity section away from the document end. ", count: 30) + "\n\n"
        try send(.threadMessageSent, messageID: "qa-stream-reply", text: sent)
        let streaming = Task { @MainActor in
            var number = 0
            while !finishStreaming {
                try Task.checkCancellation()
                let chunk = "word\(number) "
                sent += chunk
                try send(.threadMessageSent, messageID: "qa-stream-reply", text: chunk)
                number += 1
                try await Task.sleep(for: .milliseconds(120))
            }
            var binding = store.cachedThread(for: a.id)?.session
            binding?.activeTurnID = nil
            binding?.status = "ready"
            try send(.threadSessionStatusSet, session: binding)
        }
        defer { streaming.cancel() }
        try await Task.sleep(for: .seconds(2))
        try probe("before-scroll")
        ChatBenchmarkModel.beginSyntheticUserScroll(on: scroll)
        ChatBenchmarkModel.setContentOffsetY(staticBottom, on: scroll)
        ChatBenchmarkModel.endSyntheticUserScroll(on: scroll)
        try probe("after-scroll")
        try await Task.sleep(for: .milliseconds(300))
        try probe("after-settle")
        if request["probe"] == "true" {
            try await capture("positioned")
            try save("result", ["diagnosticComplete": true, "probes": probes])
            return
        }
        try await stage("last-activity-open")
        try await stage("last-thought-open")
        finishStreaming = true
        try await streaming.value
        try await Task.sleep(for: .milliseconds(500))
        // A completed long thought now leaves and re-enters the native reuse pool.
        // Page loading is triggered through the production scroll geometry path.
        guard let document = try transcript().documentView as? any ChatMacScrollDocument else { throw URLError(.unknown) }
        let visible = scroll.contentView.documentVisibleRect
        let range = document.rows(in: visible)
        let anchorRow = (range.location..<NSMaxRange(range)).first { document.rect(ofRow: $0).height > 1 } ?? range.location
        let anchorID = document.stableIdentity(forRow: anchorRow)
        let anchorOffset = document.rect(ofRow: anchorRow).minY - visible.minY
        let originalRows = document.numberOfRows
        var layoutChecks: [[String: Any]] = []
        @MainActor func checkLayout(_ name: String) throws {
            let current = try transcript()
            guard let doc = current.documentView as? any ChatMacScrollDocument else { throw URLError(.unknown) }
            let rects = (0..<doc.numberOfRows).map { doc.rect(ofRow: $0) }
            try require(!rects.isEmpty, "No document rows: " + name)
            try require(rects.allSatisfy { $0.minY.isFinite && $0.height.isFinite && $0.height >= 0 }, "Invalid row geometry: " + name)
            try require(zip(rects, rects.dropFirst()).allSatisfy { $0.maxY <= $1.minY + 1 }, "Overlapping document rows: " + name)
            try require((rects.last?.maxY ?? 0) <= doc.bounds.maxY + 1, "Last row exceeds scroll extent: " + name)
            let currentVisible = current.contentView.documentVisibleRect
            let rows = doc.rows(in: currentVisible)
            try require(rows.location != NSNotFound && rows.length > 0, "Blank viewport geometry: " + name)
            let index = anchorID.flatMap { doc.row(forStableIdentity: $0) } ?? (anchorRow + doc.numberOfRows - originalRows)
            let offset = doc.rect(ofRow: index).minY - currentVisible.minY
            layoutChecks.append(["name": name, "rows": doc.numberOfRows, "height": doc.bounds.height,
                "width": currentVisible.width, "originY": currentVisible.minY, "anchorRow": index,
                "anchorID": anchorID ?? "List index adjusted by prefix row count", "anchorOffset": offset,
                "expectedAnchorOffset": anchorOffset, "following": qaScrollState.shouldFollowBottom,
                "directSubviewCount": doc.subviews.count])
            try save("layout-checks", layoutChecks)
        }
        @MainActor func scrollTo(_ y: CGFloat) async throws {
            let current = try transcript()
            ChatBenchmarkModel.beginSyntheticUserScroll(on: current)
            ChatBenchmarkModel.setContentOffsetY(y, on: current)
            ChatBenchmarkModel.endSyntheticUserScroll(on: current)
            try await Task.sleep(for: .milliseconds(800))
        }
        try checkLayout("expanded-before-reuse")
        try await capture("expanded-before-reuse")
        for page in 0..<4 {
            try await scrollTo(0)
            try checkLayout("older-page-\(page)")
            try await capture("older-page-\(page)")
        }
        try require(document.numberOfRows > originalRows, "Older-history paging did not load more rows")
        let returnedRow = anchorID.flatMap { document.row(forStableIdentity: $0) } ?? (anchorRow + document.numberOfRows - originalRows)
        try await scrollTo(document.rect(ofRow: returnedRow).minY - anchorOffset)
        try checkLayout("returned-after-reuse")
        try await capture("returned-after-reuse")
        for (index, width) in [700.0, 1280, 800, 1100].enumerated() {
            window.setContentSize(NSSize(width: width, height: 800))
            try await Task.sleep(for: .milliseconds(800))
            try checkLayout("resized-\(index)")
            try await capture("resized-\(index)")
        }
        try await scrollTo(document.bounds.maxY)
        try checkLayout("bottom-reachable")
        try await capture("bottom-reachable")
        try require(store.selectedThread?.timeline.last?.message?.text == sent, "Final source differs")
        try require(store.selectedThread?.latestTurn?.state == "completed", "Turn did not complete")
        let finalIDs = store.selectedThread?.timeline.map { $0.message?.id ?? $0.item?.id ?? "missing" } ?? []
        try require(Array(finalIDs.prefix(originalAIDs.count)) == originalAIDs && Set(finalIDs).count == finalIDs.count, "Prior identities changed or duplicated")
        try save("result", ["passed": true, "renderer": renderer, "sourceCharacters": sent.count,
            "runtime": ProcessInfo.processInfo.operatingSystemVersionString, "stages": stages, "checkpoints": checkpoints])
    } catch {
        try save("result", ["passed": false, "error": String(describing: error), "stages": stages, "checkpoints": checkpoints])
    }
}.value
} catch {
    print("SETUP FAILURE: \(error)")
}
}
}
#endif
